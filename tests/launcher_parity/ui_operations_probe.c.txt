static void boundary(char operation) {
    int slot=(int)InterlockedIncrement((LONG*)&calls)-1;
    if(slot>=31) exit(8);
    order[slot]=operation;
    if(GetCurrentThreadId()==owner) { wrong_thread=1; return; }
    SetEvent(entered);
    if(WaitForSingleObject(release_work,5000)!=WAIT_OBJECT_0) exit(9);
}
void stop_server(void) { boundary('s'); if(!stuck) g_server_started=0; }
int start_server(void) { boundary('r'); g_server_started=!restart_fail; return !restart_fail; }
int clear_cache_file(void) { boundary('c'); cleared++; return 1; }

int themed_message(HWND h,const WCHAR *m,const WCHAR *t,UINT flags) {
    (void)h;(void)m;(void)t; dialogs++;
    if(GetCurrentThreadId()!=owner) wrong_thread=1;
    if(flags&MB_YESNO) {
        if(modal_close) ui_operation_request_close();
        if(modal_reentry) toggle_server();
    }
    return cancel && (flags&MB_YESNO) ? IDNO : IDYES;
}

HANDLE ui_operation_create_thread(LPTHREAD_START_ROUTINE work, void *arg) {
    if(fail_create) { SetLastError(ERROR_NOT_ENOUGH_MEMORY); return NULL; }
    return CreateThread(NULL,0,work,arg,0,NULL);
}

static void check(int ok,const char *s) { if(!ok) { printf("FAIL %s calls=%d wrong_thread=%d dialogs=%d cleared=%d\n",s,calls,wrong_thread,dialogs,cleared); exit(1); } }
int main(int argc,char **argv) {
    if(argc!=2) return 2;
    const char *s=argv[1]; owner=GetCurrentThreadId();
    entered=CreateEventW(NULL,TRUE,FALSE,NULL); release_work=CreateEventW(NULL,TRUE,FALSE,NULL);
    stuck=!strcmp(s,"stop-failure"); restart_fail=!strcmp(s,"restart-failure");
    fail_create=strstr(s,"create-failure")!=NULL; modal_close=!strcmp(s,"modal-close"); modal_reentry=!strcmp(s,"modal-reentry"); cancel=!strcmp(s,"cancel");
    int is_start=!strncmp(s,"start",5), is_restore=!strncmp(s,"restore",7), is_toggle=!strcmp(s,"toggle");
    if(is_toggle) toggle_server(); else if(is_start) start_translation(); else if(is_restore) restore_selected_game(); else clear_translation_cache();
    wcscpy(g_game,L"changed");
    check(!wrong_thread,"button performed blocking work on UI");
    if(fail_create || modal_close || cancel) {
        check(calls==0 && !ui_operation_busy(),"failed creation/close started work or retained busy");
        check(!fail_create || diagnostics>0,"thread creation failure lacked diagnostics");
        check(!modal_close || ui_operation_poll(),"modal close was lost");
        if(!modal_close) { check(ui_operation_reserve(),"failure/cancel did not release ownership"); ui_operation_cancel(); }
    } else {
        check(WaitForSingleObject(entered,1000)==WAIT_OBJECT_0,"worker not entered");
        int before=calls, beforeDialogs=dialogs;
        toggle_server(); clear_translation_cache(); start_translation(); restore_selected_game();
        check(calls==before && dialogs==beforeDialogs,"duplicate operations entered boundary/modal");
        for(int i=0;i<1000;i++) check(!ui_operation_poll(),"unfinished operation completed");
        int closing=strstr(s,"close")!=NULL;
        if(!strcmp(s,"window-close")) { dispatch(NULL,WM_CLOSE,0); check(!destroyed,"WM_CLOSE destroyed running operation"); }
        else if(closing) check(!ui_operation_request_close(),"close destroyed running operation");
        SetEvent(release_work);
        DWORD deadline=GetTickCount()+3000;
        int close_ready=0;
        while(ui_operation_busy()) {
            if(!strcmp(s,"window-close")) { dispatch(NULL,WM_TIMER,2); close_ready=destroyed==1; }
            else close_ready=ui_operation_poll();
            check(GetTickCount()<deadline,"completion not observed"); Sleep(1);
        }
        check(!wrong_thread,"completion dialog/card ran off UI");
        check(!strcmp(order,is_toggle?"s":is_start?"rl":is_restore?"u":stuck?"s":"scr"),"service/deploy/file operation order changed");
        if(is_toggle) check(calls==1 && dialogs==0,"toggle rejected itself");
        else if(is_start) check(calls==2 && dialogs==0,"start/deploy sequence changed");
        else if(is_restore) check(calls==1 && dialogs==(closing?1:2),"restore sequence changed");
        else if(stuck) check(cleared==0 && calls==1,"live server cache was deleted");
        else check(cleared==1 && calls==3,"stop/delete/restart sequence changed");
        check(closing ? close_ready && dialogs==(is_start?0:1) && cards==0 : cards==((is_toggle||is_start||is_restore)?0:1),"close/completion ownership failed");
    }
    CloseHandle(entered); CloseHandle(release_work); puts("PASS"); return 0;
}
